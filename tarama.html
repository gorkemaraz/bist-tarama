Python Tarama Kodu

Aşağıdaki kod, Three Inside Up taramasını çalıştırır. Backend tarafında bu scripti bir API endpointine bağladığınızda arayüz otomatik sonuçları çeker.

import yfinance as yf
import pandas as pd
from datetime import datetime, timedelta

def check_three_inside_up(df):
    if len(df) < 3:
        return False

    first, second, third = df.iloc[-3], df.iloc[-2], df.iloc[-1]

    if (
        first.isna().any() or
        second.isna().any() or
        third.isna().any()
    ):
        return False

    first_open = float(first['Open'])
    first_close = float(first['Close'])
    second_open = float(second['Open'])
    second_close = float(second['Close'])
    third_open = float(third['Open'])
    third_close = float(third['Close'])

    if (
        (first_close < first_open) and
        (second_open > first_close) and (second_close < first_open) and
        (third_close > second_open) and (third_close > first_open) and (third_open < third_close)
    ):
        return True

    return False

def find_three_inside_up_stocks():
    today = datetime.now().strftime('%Y-%m-%d')
    stocks = ["AKBNK.IS", "GARAN.IS", "THYAO.IS"]
    three_inside_up_stocks = []

    for stock in stocks:
        df = yf.download(stock, start=(datetime.now() - timedelta(days=10)).strftime('%Y-%m-%d'), end=today)
        if df.empty or df.isna().any().any():
            continue
        if check_three_inside_up(df):
            three_inside_up_stocks.append(stock)

    return three_inside_up_stocks
